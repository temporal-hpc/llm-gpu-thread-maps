def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError
    block, offset = divmod(n, 8)
    y = (offset + (offset >= 5)) // 3
    row_start = (y == 1) * 3 + (y == 2) * 5
    col_offset = offset - row_start
    x = block * 3 + col_offset * (1 + (y == 1))
    return (x, y)
