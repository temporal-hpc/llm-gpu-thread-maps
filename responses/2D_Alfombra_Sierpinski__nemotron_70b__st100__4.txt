def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input 'n' must be a non-negative integer")
    
    row = int((3 * (n ** 0.5) - (n ** (1/3))) ** (1/3))
    while True:
        col_range = (row + 1) * 3
        min_n, max_n = ((row * (row - 1)) // 2) * 3, ((row + 1) * row) // 2 * 3
        if min_n <= n < max_n:
            break
        row += 1

    col = (n - min_n) % col_range
    row = row - (col // 3)
    return (int(col), int(row))
