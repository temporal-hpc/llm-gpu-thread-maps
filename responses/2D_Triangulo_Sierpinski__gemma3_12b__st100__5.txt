def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input must be a non-negative integer.")

    x = n % 5
    y = n // 5

    if y % 5 == 0 and x > 0:
        x = 0

    if x == 1 and y > 10:
        x = 1

    if x == 1 and y == 15:
        x = 1

    if y == 16 and x > 0:
        x = 0
    
    if y == 17 and x > 0:
        x = 0

    return (x, y)
