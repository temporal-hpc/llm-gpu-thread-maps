def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input must be a non-negative integer")
    z = 0
    while (z + 1) ** 3 <= n + 1:
        z += 1
    remaining = n - ((z ** 3) - 1)
    xy_plane_size = z + 1
    y = 0
    x = 0
    for i in range(xy_plane_size):
        row_length = xy_plane_size - abs(i - z)
        if remaining < row_length:
            break
        remaining -= row_length
        y += 1
    x = remaining
    return (x, y, z)
