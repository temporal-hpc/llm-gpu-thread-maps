def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input must be a non-negative integer")
    z = 0
    while (z + 1) ** 3 <= n + 1:
        z += 1
    remainder = n - ((z ** 3) - 1)
    layer = int(remainder / (z * z))
    x = remainder % (z * z)
    y = x // z
    x %= z
    if layer == z:
        return (x, y, z)
    else:
        return (layer, y, z)
