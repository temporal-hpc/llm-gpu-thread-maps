def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError
    z = 0
    while (z + 3) * (z + 2) * (z + 1) // 6 <= n:
        z += 1
    prev = (z + 2) * (z + 1) * z // 6
    offset = n - prev
    y = 0
    while (y + 1) * (y + 2) // 2 <= offset:
        y += 1
    x = offset - y * (y + 1) // 2
    return (x, y, z)
