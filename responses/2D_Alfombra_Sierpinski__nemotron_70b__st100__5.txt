def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input 'n' must be a non-negative integer")
    
    row = int((3 * (n ** 0.5) - (n ** (1/3))) ** (1/3)) if n > 0 else 0
    col = ((3 * n + row ** 2 + (3/4) * row) ** 0.5 - (row / 2)) if n > 0 else 0
    
    offset = (row ** 2 + row) // 2
    adjusted_n = n - offset
    col += max(0, min(adjusted_n, int(row / 2)))
    
    row += max(0, adjusted_n - int(row / 2))
    
    return (int(col), int(row))
