def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input must be a non-negative integer")
    x = 0
    y = 0
    z = 0
    layer = 0
    while True:
        max_n = (2 * layer + 1) ** 3 - 1
        if n <= max_n:
            break
        layer += 1
    offset = (2 * layer) ** 3
    delta = n - ((2 * layer - 1) ** 3)
    face = delta // (4 * layer + 1)
    edge = (delta % (4 * layer + 1)) // (layer + 1)
    point = (delta % (4 * layer + 1)) % (layer + 1)
    if face == 0:
        x = layer
        y = point
        z = edge
    elif face == 1:
        x = layer - point
        y = layer
        z = edge
    elif face == 2:
        x = 0
        y = layer - point
        z = layer + edge
    else:
        x = point
        y = 0
        z = layer + edge
    return (x, y, z)
