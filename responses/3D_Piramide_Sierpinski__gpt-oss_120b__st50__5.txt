def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError
    x = y = z = 0
    factor = 1
    while n:
        d = n & 3
        if d == 1:
            x += factor
        elif d == 2:
            y += factor
        elif d == 3:
            z += factor
        n >>= 2
        factor <<= 1
    return (x, y, z)
