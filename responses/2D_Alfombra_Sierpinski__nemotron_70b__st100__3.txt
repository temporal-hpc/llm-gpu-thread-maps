def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input 'n' must be a non-negative integer.")
    
    row = int((3 * (n ** 0.5) - (n ** (1/4))) ** (2/3)) if n > 0 else 0
    col = 0
    
    while True:
        max_thread_in_row = (row * 3) + (row % 2 == 0)
        if n < max_thread_in_row:
            break
        n -= max_thread_in_row
        row += 1
    
    threads_in_partial_row = (row * 3) - ((row + 1) % 2 == 0) * row
    col = n - threads_in_partial_row // 2 if n > threads_in_partial_row // 2 else n
    
    return (col, row)
