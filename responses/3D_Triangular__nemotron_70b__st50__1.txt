def map_to_coordinates(n):
    def unravel_index(n, dims):
        coords = []
        for dim in reversed(dims):
            coords.append(n % dim)
            n //= dim
        return tuple(reversed(coords))

    dim = 0
    while (dim + 1) ** 3 <= n:
        dim += 1

    return unravel_index(n, [dim + 1, dim + 1, dim + 1])

    if not isinstance(n, int) or n < 0:
        raise ValueError("Input 'n' must be a non-negative integer")
