import math

def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError
    def C3(z):
        return (z + 3) * (z + 2) * (z + 1) // 6
    low, high = 0, 1
    while C3(high) <= n:
        high <<= 1
    while low <= high:
        mid = (low + high) // 2
        if C3(mid) <= n:
            low = mid + 1
        else:
            high = mid - 1
    z = high + 1
    prev_total = z * (z + 1) * (z + 2) // 6
    offset = n - prev_total
    disc = 8 * offset + 1
    sqrt_disc = math.isqrt(disc)
    y = (sqrt_disc - 1) // 2
    x = offset - y * (y + 1) // 2
    return (x, y, z)
