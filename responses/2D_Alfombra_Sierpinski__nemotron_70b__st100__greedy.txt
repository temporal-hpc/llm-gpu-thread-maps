def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input 'n' must be a non-negative integer")
    
    row = int((3 * (n ** 0.5) - (n ** (1/3))) ** (1/3))
    while True:
        cols_in_row = 2 if row % 3 == 0 else 3
        last_thread_in_row = row * 3 + cols_in_row - 1
        if n <= last_thread_in_row:
            break
        row += 1
    
    col = (n - (row * 3)) // (1 + (row % 3 == 0))
    return (col, row)
