def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input must be a non-negative integer")
    layer = int((n ** 0.5))
    if layer * (layer + 1) // 2 > n:
        layer -= 1
    num_in_layer = layer * (layer + 1) // 2
    pos_in_layer = n - num_in_layer
    x = pos_in_layer % (layer + 1)
    y = layer
    if pos_in_layer >= layer:
        x = layer
        y = pos_in_layer - layer + layer
    return (x, y)
