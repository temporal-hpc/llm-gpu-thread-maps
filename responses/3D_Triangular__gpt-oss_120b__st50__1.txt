def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError
    z = 0
    total = 0
    while True:
        count = (z + 1) * (z + 2) // 2
        if total + count > n:
            break
        total += count
        z += 1
    offset = n - total
    y = 0
    cum_y = 0
    while True:
        block = y + 1
        if cum_y + block > offset:
            break
        cum_y += block
        y += 1
    x = offset - cum_y
    return (x, y, z)
