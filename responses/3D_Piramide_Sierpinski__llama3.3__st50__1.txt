def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input 'n' must be a non-negative integer")
    coordinates = [0, 0, 0]
    dim = 0
    while n >= (dim + 1) ** 3 - 1:
        n -= (dim + 1) ** 3 - dim ** 3
        dim += 1
    for i in range(dim, -1, -1):
        coordinates[i % 3] = n // (i + 1)
        n %= i + 1
    return tuple(coordinates)
