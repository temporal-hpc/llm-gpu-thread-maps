def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input 'n' must be a non-negative integer.")
    layer = int(((8 * n + 1) ** 0.5 - 1) / 2)
    offset = n - (layer * (layer + 1)) // 2
    if layer == 0:
        return 0, 0
    elif offset < layer:
        return offset, layer
    else:
        offset -= layer
        if offset < layer:
            return layer, layer - offset
        else:
            offset -= layer
            return layer - offset, 0
