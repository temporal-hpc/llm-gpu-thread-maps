def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input 'n' must be a non-negative integer")

    x, y = 0, 0
    direction = 1
    steps = 1
    counter = 0

    for _ in range(n):
        for _ in range(2):
            for _ in range(steps):
                if counter == n:
                    return x, y
                x += direction
                counter += 1
            direction *= -1
        steps += 1
        y += 1

    return x, y
