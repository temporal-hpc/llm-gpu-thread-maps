def map_to_coordinates(n):
    def cantor_pairing(x, y):
        return ((x + y) * (x + y + 1)) // 2 + y

    z = int(((8 * n + 1) ** (1/3)) - 1)
    remaining = n
    for i in range(z, -1, -1):
        max_val = cantor_pairing(i, i)
        if remaining < max_val:
            break
        remaining -= max_val

    y, x = 0, z
    while True:
        pair = cantor_pairing(x, y)
        if remaining >= pair:
            remaining -= pair
            x -= 1
            y += 1
        elif x > 0:
            x -= 1
        else:
            break

    return (z, y, x)
