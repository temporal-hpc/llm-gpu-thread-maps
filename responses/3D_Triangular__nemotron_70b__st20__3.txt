def map_to_coordinates(n):
    coordinates = []
    dim = 0
    while n >= (dim + 1) ** 3 - dim ** 3:
        n -= (dim + 1) ** 3 - dim ** 3
        dim += 1
    for _ in range(3):
        coord = min(dim, n % (dim + 1))
        coordinates.append(coord)
        n //= (dim + 1)
        dim -= 1
    return tuple(sorted(coordinates))
