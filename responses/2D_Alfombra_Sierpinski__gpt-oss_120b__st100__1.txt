def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError
    offsets = [(0,0),(1,0),(2,0),(0,1),(2,1),(0,2),(1,2),(2,2)]
    x = y = 0
    power = 1
    while n:
        d = n % 8
        ox, oy = offsets[d]
        x += ox * power
        y += oy * power
        n //= 8
        power *= 3
    return (x, y)
