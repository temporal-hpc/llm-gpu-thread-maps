def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError
    block_idx, within = divmod(n, 20)
    x_block = (block_idx % 3) * 3
    y_block = (block_idx // 3) * 3
    if within < 8:
        dz = 0
        idx = within
        count = 0
        for dy in range(3):
            xs = (0, 1, 2) if dy != 1 else (0, 2)
            for dx in xs:
                if count == idx:
                    return (x_block + dx, y_block + dy, dz)
                count += 1
    elif within < 12:
        dz = 1
        idx = within - 8
        dy = 0 if idx < 2 else 2
        dx = 0 if idx % 2 == 0 else 2
        return (x_block + dx, y_block + dy, dz)
    else:
        dz = 2
        idx = within - 12
        count = 0
        for dy in range(3):
            xs = (0, 1, 2) if dy != 1 else (0, 2)
            for dx in xs:
                if count == idx:
                    return (x_block + dx, y_block + dy, dz)
                count += 1
