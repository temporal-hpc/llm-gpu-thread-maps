def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input 'n' must be a non-negative integer")
    
    row = int((3 * (n ** 0.5)) - (3 / 2) + 1)
    while True:
        max_cols = int(((row ** 2) / 3) + (row / 2))
        if n < (max_cols + ((row - 1) // 3) * 3):
            break
        row += 1
    
    col = (n - ((row - 1) // 3) * 3)
    if row % 3 == 0:
        return (col, row // 3)
    elif row % 3 == 1:
        return (2 - (col % 3), row // 3)
    else:
        return ((col + 1) // 3, row // 3)
