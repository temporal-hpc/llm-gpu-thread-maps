def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input must be a non-negative integer")
    z = 0
    while (z + 1) ** 3 <= n + 1:
        z += 1
    n -= sum([(i + 1) ** 2 for i in range(z)])
    y = 0
    while (y + 1) * (z + 1) <= n + 1:
        y += 1
    n -= sum([z + 1 for _ in range(y)])
    x = n
    return (x, y, z)
