def map_to_coordinates(n):
    def cantor_pairing(x, y):
        return ((x + y) * (x + y + 1)) // 2 + y

    z = int(((8 * n + 1) ** (1/3)))
    while True:
        if ((z - 1) / 6) ** 3 <= n and n < (z ** 3) / 6:
            break
        z += 1

    m = (z ** 3) // 6 - n
    x, y = 0, 0
    for i in range(z-1, -1, -1):
        j = min(i, int(m))
        m -= j
        if j == i:
            x += 1 << (z - 1 - i)
        else:
            y += 1 << (z - 1 - i)

    return (x, y, z-1)
