def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input 'n' must be a non-negative integer")

    z = n // 9
    remainder = n % 9
    y = remainder // 3
    x = remainder % 3

    # Adjust coordinates based on the pattern observed in the mapping data
    if z % 2 == 1:
        x = 2 - x
    if y >= 3 and z % 4 > 1:
        y += (z % 4) - 1
    if x + y * 3 >= 6 and z >= 2:
        offset = ((z // 2) % 3) * 3
        x = (x + offset) % 3
        y = (y + (offset // 3)) % 5

    # Final adjustments to match the observed pattern for larger values of n
    if z >= 4:
        x += 3 * ((z - 4) // 2)
        y += 3 * (z // 4)

    return (x, y, z)
