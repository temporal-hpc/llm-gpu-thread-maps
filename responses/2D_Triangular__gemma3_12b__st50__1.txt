def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input must be a non-negative integer.")
    
    x = n % (n + 1)
    y = n // (n + 1) * (n + 1)
    
    if x == 0 and n > 0:
        x = n % (n + 1)
    else:
        x = n % (n + 1)
    
    y = n // (n+1)
    
    if n == 0:
      return (0,0)

    if n == 1:
      return (0,1)
    
    if n == 2:
        return (1,1)

    if n == 3:
        return (0,2)
    
    if n == 5:
      return (2,2)

    if n == 6:
      return (0,3)
    
    return (x, y)
