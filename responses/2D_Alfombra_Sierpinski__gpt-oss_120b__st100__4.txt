def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError
    block_index, p = divmod(n, 8)
    block_row, block_col = divmod(block_index, 3)
    origin_x = block_col * 3
    origin_y = block_row * 3
    dx, dy = ((0,0),(1,0),(2,0),(0,1),(2,1),(0,2),(1,2),(2,2))[p]
    return (origin_x + dx, origin_y + dy)
