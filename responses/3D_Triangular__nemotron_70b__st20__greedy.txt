def map_to_coordinates(n):
    def cantor_pairing(x, y):
        return ((x + y) * (x + y + 1)) // 2 + y

    z = int(((8 * n + 1) ** (1/3)))
    while (((z - 1) / 2) ** 3) + ((z - 1) / 2) <= n:
        z += 1
    z -= 1
    n -= (((z - 1) / 2) ** 3) + ((z - 1) / 2)
    y = int(((-3 + (81 + 12 * n) ** (1/2)) / 6))
    while cantor_pairing(y, z) > n:
        y -= 1
    x = n - cantor_pairing(y, z)
    return (x, y, z)
