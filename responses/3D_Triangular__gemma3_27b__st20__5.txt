def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input must be a non-negative integer.")

    z = 0
    while (z + 1) * (z + 1) * (z + 1) <= n:
        z += 1

    n_reduced = n - (z - 1) * (z - 1) * (z - 1)

    y = 0
    while (y + 1) * (y + 1) <= n_reduced:
        y += 1

    n_reduced_2 = n_reduced - (y - 1) * (y - 1)

    x = n_reduced_2

    return (x, y - 1, z - 1)
