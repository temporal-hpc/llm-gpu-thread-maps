def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input 'n' must be a non-negative integer")
    
    x, y = 0, 0
    direction = 1
    
    for _ in range(n):
        if (x == y and x % 2 == 1) or (x != y and x < y and (y - x) % 2 == 1 and x + y >= n):
            direction = -direction
        x += direction
    
    while x < 0:
        x, y = x + 1, y + 1
    
    return (x, y)
