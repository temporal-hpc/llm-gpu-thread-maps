def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input 'n' must be a non-negative integer.")
    d = 0
    while (d + 1) ** 3 <= n + 1:
        d += 1
    n -= d ** 3
    for i in range(d):
        if n < (i + 1) * (2 * d - 2 * i + 1):
            break
    n -= i * (2 * d - 2 * i + 1)
    p = 0
    while n >= 2 * d - 2 * i - 4 * p:
        n -= 2 * d - 2 * i - 4 * p
        p += 1
    q = (n + p) % (d - i - p)
    r = (n + p) // (d - i - p)
    if r == 0 and q == d - i - p:
        return (i, p, d)
    elif r == 0:
        return (i, d, p + q)
    else:
        return (p + q, d, i + r)
