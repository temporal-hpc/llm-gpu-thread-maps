def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input 'n' must be a non-negative integer.")
    x, y, z = 0, 0, 0
    while n >= 7:
        remainder = n % 7
        n = n // 7
        if remainder == 1:
            x += 1
        elif remainder == 2:
            y += 1
        elif remainder == 3:
            z += 1
        elif remainder == 4:
            x += 1
            y += 1
        elif remainder == 5:
            x += 1
            z += 1
        elif remainder == 6:
            y += 1
            z += 1
    if n == 1:
        x += 1
    elif n == 2:
        y += 1
    elif n == 3:
        z += 1
    elif n == 4:
        x += 1
        y += 1
    elif n == 5:
        x += 1
        z += 1
    elif n == 6:
        y += 1
        z += 1
    return x, y, z
