def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("n must be a non‑negative integer")
    x = y = z = 0
    shift = 0
    m = n
    while m:
        b0 = m & 1
        b1 = (m >> 1) & 1
        if b0 == 1 and b1 == 0:
            x |= 1 << shift
        elif b0 == 0 and b1 == 1:
            y |= 1 << shift
        elif b0 == 1 and b1 == 1:
            z |= 1 << shift
        m >>= 2
        shift += 1
    return (x, y, z)
