def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input 'n' must be a non-negative integer")

    z = int((3 * (n ** (1/2)) - 3 * (n ** (1/3)) + 1) / 2)
    while True:
        layer_size = (z + 1) * (z + 2) // 2
        if n < layer_size + (z + 2):
            break
        z += 1

    n -= sum((i + 1) for i in range(z))
    y, x = divmod(n - 1, z + 1)
    return (x, min(y, z - y), z)
