def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("n must be a non‑negative integer")
    # find z such that C(z-1) <= n < C(z), where C(k) = (k+1)(k+2)(k+3)//6
    low, high = 0, 1
    while (high + 1) * (high + 2) * (high + 3) // 6 <= n:
        high *= 2
    while low < high:
        mid = (low + high) // 2
        if (mid + 1) * (mid + 2) * (mid + 3) // 6 <= n:
            low = mid + 1
        else:
            high = mid
    z = low
    prev_c = z * (z + 1) * (z + 2) // 6  # C(z-1)
    offset = n - prev_c  # 0‑based index within layer z
    # find y such that cumulative_y > offset, cumulative_y = (y+1)(y+2)//2
    low, high = 0, z
    while low < high:
        mid = (low + high) // 2
        if (mid + 1) * (mid + 2) // 2 <= offset:
            low = mid + 1
        else:
            high = mid
    y = low
    prev_c_y = y * (y + 1) // 2  # cumulative up to y-1
    x = offset - prev_c_y
    return (x, y, z)
