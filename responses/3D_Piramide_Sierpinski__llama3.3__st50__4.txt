def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input must be a non-negative integer")
    coordinates = [0, 0, 0]
    i = 0
    while n >= 0:
        coordinates[i % 3] += 1
        n -= (2 ** (i // 3)) * ((i % 3) + 1)
        if n < 0:
            coordinates[i % 3] -= 1
            break
        i += 1
    return tuple(coordinates)
