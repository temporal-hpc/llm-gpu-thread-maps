def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input 'n' must be a non-negative integer.")
    if n == 0:
        return (0, 0, 0)
    x, y, z = 0, 0, 0
    base = 1
    while n > 0:
        digit = n % 8
        n //= 8
        if digit == 0:
            x += base * 0
            y += base * 0
            z += base * 1
        elif digit == 1:
            x += base * 1
            y += base * 0
            z += base * 0
        elif digit == 2:
            x += base * 0
            y += base * 1
            z += base * 0
        elif digit == 3:
            x += base * 0
            y += base * 0
            z += base * 1
        elif digit == 4:
            x += base * 0
            y += base * 0
            z += base * 0
        elif digit == 5:
            x += base * 0
            y += base * 0
            z += base * 0
        elif digit == 6:
            x += base * 0
            y += base * 0
            z += base * 0
        elif digit == 7:
            x += base * 0
            y += base * 0
            z += base * 0
        base *= 8
    return (x, y, z)
