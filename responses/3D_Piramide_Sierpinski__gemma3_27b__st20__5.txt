    if not isinstance(n, int) or n < 0:
        raise ValueError("Input must be a non-negative integer.")

    z = n // 4
    rem = n % 4

    if rem == 0:
        x = 0
        y = 0
    elif rem == 1:
        x = 1
        y = 0
    elif rem == 2:
        x = 0
        y = 1
    else:
        x = 0
        y = 0

    if z % 2 == 0:
        x_final = x
        y_final = y
        z_final = z // 2
    else:
        x_final = y
        y_final = x
        z_final = z // 2
