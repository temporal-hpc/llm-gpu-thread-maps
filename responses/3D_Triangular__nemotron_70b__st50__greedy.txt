def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input 'n' must be a non-negative integer")

    z = int((3 * (n ** (1/2)) - 3 * n + 2) ** (1/3))
    while True:
        layer_size = ((z + 1) * (z + 2)) // 2
        if n < layer_size * (z + 3) // 2:
            break
        z += 1

    offset = ((z * (z + 1)) // 2) * (z + 2) // 2
    n -= offset

    y = min(z, int((n * 2) ** (1/2)))
    while True:
        row_size = y + 1 if y < z else z + 1
        if n < row_size:
            break
        n -= row_size
        y += 1

    x = n if y < z else z
    return (x, y, z)
