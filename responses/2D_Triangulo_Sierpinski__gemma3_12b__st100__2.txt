def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input must be a non-negative integer.")

    if n == 0:
        return (0, 0)

    x = n % 18
    y = n // 18

    if x == 0 and y == 0:
        return (0, 0)
    
    if x == 0 and y % 18 == 0:
        return (0,y)
    
    if x == 0:
        return (0, y)

    if y == 0:
        return (x,0)

    if x == 1 and y == 1:
        return (0,1)
    
    if x == 17 and y == 17:
        return (16,16)
    
    if y == 17 and x ==1:
        return (0,17)

    if x >= 16 and y >= 16:
      return (16, 16)

    if y == 17:
        return (x,y)
    
    if x == 17:
        return (x, y)
    

    return (x, y)
