def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError
    cum = 0
    y = 0
    while True:
        cnt = 1 << bin(y).count('1')
        if n < cum + cnt:
            break
        cum += cnt
        y += 1
    offset = n - cum
    x = 0
    bit_idx = 0
    pos = 0
    temp = y
    while temp:
        if temp & 1:
            if offset & (1 << bit_idx):
                x |= 1 << pos
            bit_idx += 1
        pos += 1
        temp >>= 1
    return (x, y)
