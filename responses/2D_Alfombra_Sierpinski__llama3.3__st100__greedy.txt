def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input 'n' must be a non-negative integer")
    layer = int((3 * (n ** 0.5) - 3) ** 2 + 1)
    pos_in_layer = n - ((layer - 4) // 6) ** 2
    side_length = (layer - 1) // 3
    if pos_in_layer < side_length:
        return pos_in_layer, layer // 3
    elif pos_in_layer < 2 * side_length:
        return layer // 3, side_length - (pos_in_layer - side_length)
    else:
        return layer // 3 - (pos_in_layer - 2 * side_length), side_length - (layer // 3)
