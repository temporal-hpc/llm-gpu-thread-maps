def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input must be a non-negative integer.")

    z = 0
    while (z + 1) * (z + 1) * (z + 1) <= n:
        z += 1
    z -= 1

    n_offset = z * z * z
    
    x = 0
    while (x + 1) * (x + 1) <= (n - n_offset):
        x += 1
    x -= 1

    n_offset_x = x * x
    
    y = n - n_offset - n_offset_x

    return (x, y, z)
