def map_to_coordinates(n):
    coords = [0, 0, 0]
    axis = 0
    increment_axis = False
    i = 1
    
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input 'n' must be a non-negative integer.")
    
    if n == 0:
        return tuple(coords)
    
    while n > 0:
        if increment_axis:
            axis = (axis + 1) % 3
            increment_axis = False
        else:
            coords[axis] += i
            n -= 1
            if n == 0:
                break
            if coords[axis] ** 2 + coords[(axis+1)%3] ** 2 + coords[(axis+2)%3] ** 2 == (i * (i+1)) // 2 * 2:
                increment_axis = True
        i += 1
    
    return tuple(coords)
