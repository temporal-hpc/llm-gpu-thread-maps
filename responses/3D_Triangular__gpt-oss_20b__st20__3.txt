def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input must be a non‑negative integer")
    def C(z):
        return (z + 1) * (z + 2) * (z + 3) // 6
    low, high = 0, 1
    while C(high) <= n:
        high <<= 1
    while low < high:
        mid = (low + high) // 2
        if C(mid) > n:
            high = mid
        else:
            low = mid + 1
    z = low
    prev_z = z * (z + 1) * (z + 2) // 6
    offset = n - prev_z
    low_y, high_y = 0, z
    while low_y < high_y:
        mid = (low_y + high_y) // 2
        if (mid + 1) * (mid + 2) // 2 > offset:
            high_y = mid
        else:
            low_y = mid + 1
    y = low_y
    prev_y = y * (y + 1) // 2
    x = offset - prev_y
    return (x, y, z)
