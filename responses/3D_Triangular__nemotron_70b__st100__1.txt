def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input 'n' must be a non-negative integer")
    
    z = int((3 * (n ** (1/2)) - 1) / 2)
    while True:
        layer_size = (z + 1) * (z + 2) // 2
        if n < layer_size + (z + 2):
            break
        z += 1
    
    n -= (z * (z + 1)) // 2
    x, y = 0, z
    while n > z:
        n -= z
        x, y = y, x - 1
    return (x + n, y - n, z)
