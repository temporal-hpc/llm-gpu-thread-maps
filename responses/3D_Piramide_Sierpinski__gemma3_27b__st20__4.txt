def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input must be a non-negative integer.")

    z = n // 4
    rem = n % 4

    if rem == 0:
        x = 0
        y = 0
        z = z
    elif rem == 1:
        x = 1
        y = 0
        z = z
    elif rem == 2:
        x = 0
        y = 1
        z = z
    else:
        x = 0
        y = 0
        z = z + 1

    if z == 0 and x == 0 and y == 0:
        return (0, 0, 0)
    if z == 0 and x == 1 and y == 0:
        return (1, 0, 0)
    if z == 0 and x == 0 and y == 1:
        return (0, 1, 0)
    if z == 0 and x == 0 and y == 0:
        return (0, 0, 1)
    
    if n == 4:
        return (2, 0, 0)
    if n == 5:
        return (3, 0, 0)
    if n == 6:
        return (2, 1, 0)
    if n == 7:
        return (2, 0, 1)
    if n == 8:
        return (0, 2, 0)
    if n == 9:
        return (1, 2, 0)
    if n == 10:
        return (0, 3, 0)
    if n == 11:
        return (0, 2, 1)
    if n == 12:
        return (0, 0, 2)
    if n == 13:
        return (1, 0, 2)
    if n == 14:
        return (0, 1, 2)
    if n == 15:
        return (0, 0, 3)
    if n == 16:
        return (4, 0, 0)
    if n == 17:
        return (5, 0, 0)
    if n == 18:
        return (4, 1, 0)
    if n == 19:
        return (4, 0, 1)
        
    x = n // 4
    y = (n % 4) // 2
    z = (n % 4) % 2

    return (x, y, z)
