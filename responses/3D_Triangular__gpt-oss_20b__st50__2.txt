def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("n must be a non‑negative integer")
    def S(z):
        return (z + 1) * (z + 2) * (z + 3) // 6
    lo, hi = 0, 1
    while S(hi) <= n:
        hi <<= 1
    while lo < hi:
        mid = (lo + hi) // 2
        if S(mid) > n:
            hi = mid
        else:
            lo = mid + 1
    z = lo
    prev = S(z - 1) if z > 0 else 0
    offset = n - prev
    lo, hi = 0, z
    while lo < hi:
        mid = (lo + hi) // 2
        if (mid + 1) * (mid + 2) // 2 > offset:
            hi = mid
        else:
            lo = mid + 1
    y = lo
    x = offset - y * (y + 1) // 2
    return (x, y, z)
