import math

def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("n must be a non‑negative integer")
    def S(z):
        return (z + 1) * (z + 2) * (z + 3) // 6
    low, high = 0, 1
    while S(high) <= n:
        high <<= 1
    while low < high:
        mid = (low + high) >> 1
        if S(mid) > n:
            high = mid
        else:
            low = mid + 1
    z = low
    prev = S(z - 1) if z > 0 else 0
    offset = n - prev
    y = (math.isqrt(8 * offset + 1) - 1) // 2
    x = offset - y * (y + 1) // 2
    return (x, y, z)
