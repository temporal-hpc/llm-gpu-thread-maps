def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError
    cum = 0
    y = 0
    while True:
        cnt = 1 << y.bit_count()
        if cum + cnt > n:
            offset = n - cum
            bits = []
            temp = y
            pos = 0
            while temp:
                if temp & 1:
                    bits.append(pos)
                temp >>= 1
                pos += 1
            x = 0
            for i, b in enumerate(bits):
                if (offset >> i) & 1:
                    x |= 1 << b
            return (x, y)
        cum += cnt
        y += 1
