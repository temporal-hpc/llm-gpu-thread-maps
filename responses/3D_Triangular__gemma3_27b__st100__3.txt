def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input must be a non-negative integer.")

    z = 0
    while (z + 1) * (z + 1) * (z + 1) <= n:
        z += 1
    z -= 1

    n_offset = z * z * z
    n_relative = n - n_offset

    y = 0
    while (y + 1) * (y + 1) <= n_relative:
        y += 1
    y -= 1

    y_offset = y * y
    n_relative_offset = n_relative - y_offset

    x = n_relative_offset

    return (x, y, z)
