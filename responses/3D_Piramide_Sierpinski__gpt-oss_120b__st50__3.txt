def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError
    x = y = z = 0
    power = 1
    while n:
        d = n & 3
        if d == 1:
            x += power
        elif d == 2:
            y += power
        elif d == 3:
            z += power
        n >>= 2
        power <<= 1
    return (x, y, z)
