def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input must be a non-negative integer.")

    z = 0
    while (z + 1) * (z + 1) * (z + 1) <= n:
        z += 1
    z -= 1

    n_prime = n - z * z * z

    y = 0
    while (y + 1) * (y + 1) <= n_prime:
        y += 1
    y -= 1

    n_double_prime = n_prime - y * y

    x = n_double_prime

    return (x, y, z)
