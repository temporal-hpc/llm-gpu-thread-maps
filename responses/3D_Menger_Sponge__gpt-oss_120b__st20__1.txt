def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError
    cnt = 0
    for z in range(3):
        for y in range(3):
            for x in range(3):
                if ((x in (0, 2)) + (y in (0, 2)) + (z in (0, 2))) >= 2:
                    if cnt == n:
                        return (x, y, z)
                    cnt += 1
    raise ValueError
