def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input must be a non-negative integer.")

    x = (n % 4) // 2
    y = n // 4
    
    if n % 4 == 1 or n % 4 == 3:
        x = 1
    if n % 4 == 2:
        x = 2
    if n % 4 == 3:
        x = 3

    if n % 4 == 0:
        x = 0
    
    if n == 10:
      x = 4
    if n == 13:
      x = 1
    if n == 14:
      x = 5
    if n == 16:
      x = 2
    if n == 17:
      x = 4
    if n == 18:
      x = 6
    if n == 19:
      x = 0
    

    return (x, y)
