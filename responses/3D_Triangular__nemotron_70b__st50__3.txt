def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input 'n' must be a non-negative integer")
    
    dim = 3
    coords = [0] * dim
    remaining = n
    
    for i in range(dim-1, -1, -1):
        if i == dim-1:
            max_val = int((remaining ** (1. / i)))
        else:
            max_val = int(((remaining + 1) ** (1. / i)) - 1)
        
        coords[i] = min(max_val, remaining)
        remaining -= coords[i]
    
    # Adjust coordinates to fit the observed pattern
    for i in range(1, dim):
        coords[i] += coords[i-1]
    
    return tuple(coords)
