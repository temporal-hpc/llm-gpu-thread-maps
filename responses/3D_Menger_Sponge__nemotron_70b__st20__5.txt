def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input 'n' must be a non-negative integer")
    
    z = n // 9
    remainder = n % 9
    
    if remainder in (1, 4, 7):  
        x = 2
        y = (remainder - 1) // 3
    else:  
        x = (3 * remainder + 2) % 3
        y = (remainder - x) // 2 if remainder >= 3 else 0
    
    return (x, y, z)
