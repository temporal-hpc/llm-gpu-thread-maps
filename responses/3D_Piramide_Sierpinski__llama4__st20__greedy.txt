def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input 'n' must be a non-negative integer.")
    x, y, z = 0, 0, 0
    while n >= 6:
        x += 1
        n -= 6
    if n >= 3:
        y += 1
        n -= 3
    if n >= 1:
        z += 1
        n -= 1
    if n == 0:
        return (x, y, z)

        return (x, y, z + 1)
    elif n == 2:

def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input 'n' must be a non-negative integer.")
    x = n // 6
    remainder = n % 6
    y = remainder // 3
    z = remainder % 3
    return (x, y, z)
