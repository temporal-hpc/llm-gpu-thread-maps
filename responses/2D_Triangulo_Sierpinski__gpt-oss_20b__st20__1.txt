def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError
    total = 0
    y = 0
    while True:
        subsets = 1 << bin(y).count('1')
        if total + subsets > n:
            break
        total += subsets
        y += 1
    index = n - total
    y_val = y
    pos = 0
    x = 0
    mask = index
    while y_val:
        if y_val & 1:
            if mask & 1:
                x |= 1 << pos
            mask >>= 1
        y_val >>= 1
        pos += 1
    return (x, y)
