def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input 'n' must be a non-negative integer")
    
    z = n // 9
    remainder = n % 9
    y = remainder // 3
    x = remainder % 3
    
    if (x + y + z) % 2 != 0:
        x = 2 - x
    
    offset = (z // 3) * 3
    x += offset
    if (y >= z % 3):
        x += (z % 3) * 3
        y -= z % 3
    
    return (x, y, z)
