def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError
    y = 0
    r = n
    while True:
        size = 1 << bin(y).count('1')
        if r < size:
            offset = r
            x = 0
            bit = 0
            yy = y
            while yy:
                low = yy & -yy
                pos = low.bit_length() - 1
                if offset & (1 << bit):
                    x |= 1 << pos
                bit += 1
                yy ^= low
            return (x, y)
        r -= size
        y += 1
