def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError
    z = 0
    while (z + 1) * (z + 2) * (z + 3) // 6 <= n:
        z += 1
    prev = z * (z + 1) * (z + 2) // 6
    t = n - prev
    y = 0
    while (y + 1) * (y + 2) // 2 <= t:
        y += 1
    x = t - y * (y + 1) // 2
    return (x, y, z)
