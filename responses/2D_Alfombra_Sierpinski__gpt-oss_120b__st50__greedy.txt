def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError
    count = 0
    y = 0
    while True:
        x = 0
        while True:
            if not (x % 3 == 1 and y % 3 == 1) and not ((x // 3) % 2 == 1 and (y // 3) % 2 == 1):
                if count == n:
                    return (x, y)
                count += 1
            x += 1
            if count > n and x > n * 3:
                break
        y += 1
