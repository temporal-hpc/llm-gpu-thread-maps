def map_to_coordinates(n):
    z = int((n ** (1/3))) if n > 0 else 0
    remainder = n - (z ** 3)
    y = int(((remainder ** (1/2)))) if remainder > 0 else 0
    x = remainder - (y * ((y + 1) // 2))
    while x < 0:
        z -= 1
        remainder = n - (z ** 3)
        y = int(((remainder ** (1/2)))) if remainder > 0 else 0
        x = remainder - (y * ((y + 1) // 2))
    while x >= (y + 1):
        y += 1
        x -= y
    return x, y, z
