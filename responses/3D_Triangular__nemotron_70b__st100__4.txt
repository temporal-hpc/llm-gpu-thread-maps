def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input 'n' must be a non-negative integer")

    z = int((3 * (n ** 2) + n) ** 0.5) // 3 + min(1, n)
    while ((3 * (z ** 2) - z) // 2) <= n:
        z += 1
    z -= 1

    y = min(z, n - ((3 * (z ** 2) - z) // 2))
    while (n - ((3 * (z ** 2) - z) // 2)) - y >= z - y + 1:
        y += 1
    y -= 1

    x = n - ((3 * (z ** 2) - z) // 2) - y
    return (x, y, z)
