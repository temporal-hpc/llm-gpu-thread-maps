def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input must be a non-negative integer.")

    z = n // 4
    rem = n % 4

    if rem == 0:
        x = 0
        y = 0
    elif rem == 1:
        x = 1
        y = 0
    elif rem == 2:
        x = 0
        y = 1
    else:
        x = 0
        y = 0

    if z % 2 == 0:
        x_final = x
        y_final = y
        z_final = z // 2
    else:
        x_final = y
        y_final = x
        z_final = z // 2

    if z >= 4:
        x_final = x_final + (z // 2) * 2
        y_final = y_final + (z // 2) * 2

    if z == 1:
        x_final = x
        y_final = y
        z_final = 0
    elif z == 2:
        x_final = x
        y_final = y
        z_final = 1
    elif z == 3:
        x_final = x
        y_final = y
        z_final = 1
    elif z == 4:
        x_final = x
        y_final = y
        z_final = 2
    elif z == 5:
        x_final = x
        y_final = y
        z_final = 2
    elif z == 6:
        x_final = x
        y_final = y
        z_final = 3
    elif z == 7:
        x_final = x
        y_final = y
        z_final = 3

    if n == 16:
        return (4, 0, 0)
    if n == 17:
        return (5, 0, 0)
    if n == 18:
        return (4, 1, 0)
    if n == 19:
        return (4, 0, 1)

    return (x_final, y_final, z_final)
