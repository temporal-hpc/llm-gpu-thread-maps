def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input 'n' must be a non-negative integer")
    
    row = int(((8 * n + 1) ** 0.5 - 1) / 2)
    col = n - (row * (row + 1)) // 2
    
    while col > row:
        row += 1
        col -= 1
    
    return (col, row)
