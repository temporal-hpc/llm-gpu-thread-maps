def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input 'n' must be a non-negative integer")
    
    z = n // 9 - (n // 18) * 2
    remainder = n % 9
    y = remainder // 3
    x = remainder % 3 + (remainder // 3) * 2 - ((remainder % 3) * (remainder // 3))
    
    if (x, y, z) == (4, 0, 0):
        x, y, z = 3, 0, 0
        for _ in range((n - 8) // 9):
            if y == 2:
                x += 1
                y = 0
            else:
                y += 1
            if (x, y, z) in [(4, 1, 0), (4, 2, 0)]:
                x -= 1
    
    elif x > 3 or y > 2:
        offset = ((n - 8) // 9) * 3
        x = 6 + (x - 4)
        if y == 3: 
            y, z = 0, 1
        elif y == 4: 
            y, z = 1
        else: 
            y, z = 2, 1
        for _ in range(offset):
            if y == 2 and x > 6:
                x += 1
                y = 0
            elif x > 8:
                x -= 3
                y += 1
            else:
                x += 1
    
    return (x, y, z)
