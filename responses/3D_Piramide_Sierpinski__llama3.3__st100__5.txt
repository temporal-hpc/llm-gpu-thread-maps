def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input must be a non-negative integer")
    x = 0
    y = 0
    z = 0
    i = 1
    while True:
        max_val = (i * (i + 1)) // 2 - 1
        if n <= max_val:
            break
        n -= max_val + 1
        i += 1
    layer_size = i
    offset = (layer_size * (layer_size - 1)) // 2
    n += offset
    side_length = layer_size
    while True:
        if n < side_length:
            x = n
            break
        n -= side_length
        side_length -= 1
        if n < side_length:
            y = n
            break
        n -= side_length
        side_length -= 1
        z = n
    return (x, y, z)
