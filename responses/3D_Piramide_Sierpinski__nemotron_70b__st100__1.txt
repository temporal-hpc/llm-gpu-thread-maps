def map_to_coordinates(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Input 'n' must be a non-negative integer")
    
    def cantor_pairing(x, y):
        return ((x + y) * (x + y + 1)) // 2 + y
    
    z = 0
    while True:
        max_cantor = ((z + n) * (z + n + 1)) // 2 + n
        if max_cantor >= n:
            break
        n -= max_cantor - ((z * (z + 1)) // 2)
        z += 1
    
    y = 0
    while True:
        cantor = ((z + y) * (z + y + 1)) // 2 + y
        if cantor > n:
            break
        n -= cantor - ((z * (z + 1)) // 2)
        y += 1
    
    x = n - ((z + y) * (z + y + 1)) // 2 + y
    return (x, y, z)
